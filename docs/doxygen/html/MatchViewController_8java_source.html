125 private MatchResultRequestDTO toResultRequest(List<MatchGoal> goals) {
126 MatchResultRequestDTO resultRequest = new MatchResultRequestDTO();
127 List<MatchResultRequestDTO.GoalDTO> goalDTOs = new ArrayList<>();
128
129 for (MatchGoal goal : goals) {
130 MatchResultRequestDTO.GoalDTO goalDTO = new MatchResultRequestDTO.GoalDTO();
131 goalDTO.setTeamId(goal.getTeam().getId());
132 goalDTO.setMinute((int) goal.getMinute());
133 goalDTO.setStoppageMinute(
134 goal.getStoppageMinute() == null ? null : (int) goal.getStoppageMinute()
135 );
136 goalDTOs.add(goalDTO);
137 }
138
139 resultRequest.setGoals(goalDTOs);
140 return resultRequest;
141 }
