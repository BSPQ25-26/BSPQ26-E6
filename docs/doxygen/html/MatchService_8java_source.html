151 @Transactional
152 public String registerResult(Long id, MatchResultRequestDTO request) {
153 if (request == null || request.getGoals() == null) {
154 throw new IllegalArgumentException("Validation Error: Goals are required.");
155 }
156
157 Match match = matchRepository.findById(id)
158 .orElseThrow(() -> new IllegalArgumentException("Match not found with id: " + id));
159
160 List<MatchGoal> goals = new ArrayList<>();
161 int leftScore = 0;
162 int rightScore = 0;
163
164 for (MatchResultRequestDTO.GoalDTO goalDTO : request.getGoals()) {
165 Team scoringTeam = resolveScoringTeam(match, goalDTO);
166
167 if (Objects.equals(scoringTeam.getId(), match.getLeftTeam().getId())) {
168 leftScore++;
169 } else {
170 rightScore++;
171 }
172
173 MatchGoal goal = new MatchGoal();
174 goal.setMatch(match);
175 goal.setTeam(scoringTeam);
176 goal.setMinute(goalDTO.getMinute().shortValue());
177 goal.setStoppageMinute(toShort(goalDTO.getStoppageMinute()));
178 goals.add(goal);
179 }
180
181 matchGoalRepository.deleteByMatchId(id);
182 matchGoalRepository.saveAll(goals);
183
184 match.setLeftScore((short) leftScore);
185 match.setRightScore((short) rightScore);
186 match.setFinished(true);
187 matchRepository.save(match);
188
189 return "Result registered for match " + id + ": " + leftScore + " - " + rightScore;
190 }

322 private Team resolveScoringTeam(Match match, MatchResultRequestDTO.GoalDTO goalDTO) {
323 if (goalDTO == null || goalDTO.getTeamId() == null || goalDTO.getMinute() == null) {
324 throw new IllegalArgumentException("Validation Error: Every goal must include team ID and minute.");
325 }
326
327 if (goalDTO.getMinute() < 1 || goalDTO.getMinute() > 120) {
328 throw new IllegalArgumentException("Validation Error: Goal minute must be between 1 and 120.");
329 }
330
331 if (goalDTO.getStoppageMinute() != null
332 && (goalDTO.getStoppageMinute() < 0 || goalDTO.getStoppageMinute() > 30)) {
333 throw new IllegalArgumentException("Validation Error: Stoppage minute must be between 0 and 30.");
334 }
335
336 if (Objects.equals(goalDTO.getTeamId(), match.getLeftTeam().getId())) {
337 return match.getLeftTeam();
338 }
339
340 if (Objects.equals(goalDTO.getTeamId(), match.getRightTeam().getId())) {
341 return match.getRightTeam();
342 }
343
344 throw new IllegalArgumentException("Validation Error: Scoring team must belong to the match.");
345 }
